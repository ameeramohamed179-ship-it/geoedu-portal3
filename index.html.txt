// ========== Audit Log ==========
const AuditLog = {
  KEY: 'otp_audit_log',
  MAX: 50,
  _read() { try { return JSON.parse(localStorage.getItem(this.KEY) || '[]'); } catch { return []; } },
  _write(list) { localStorage.setItem(this.KEY, JSON.stringify(list.slice(-this.MAX))); },
  add(record) {
    const list = this._read();
    list.push({ otp: record.otp, hash: record.hash, counter: record.counter, issuedAt: record.issuedAt, status: 'pending' });
    this._write(list);
  },
  markVerified(otp) {
    const list = this._read();
    const e = list.find(x => x.otp === otp);
    if (e) { e.status = 'verified'; e.verifiedAt = new Date().toISOString(); }
    this._write(list);
  },
  markExpired(otp) {
    const list = this._read();
    const e = list.find(x => x.otp === otp);
    if (e) e.status = 'expired';
    this._write(list);
  },
  getRecent(n = 5) { return this._read().slice(-n).reverse(); },
};

// ========== OTP Generator (unique, non-repeatable) ==========
async function generateUniqueOTP(userId) {
  const randomBytes = new Uint8Array(32);
  crypto.getRandomValues(randomBytes);
  const randomHex = Array.from(randomBytes).map(b => b.toString(16).padStart(2, '0')).join('');
  const timestamp = Date.now().toString(36);
  const counter = parseInt(localStorage.getItem('otp_counter') || '0', 10) + 1;
  localStorage.setItem('otp_counter', counter.toString());

  const fingerprint = [navigator.userAgent, `${screen.width}x${screen.height}`,
    Intl.DateTimeFormat().resolvedOptions().timeZone, navigator.language].join('|');

  const seed = `${userId}|${timestamp}|${counter}|${fingerprint}|${randomHex}`;
  const hashBuffer = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(seed));
  const hashArray = Array.from(new Uint8Array(hashBuffer));
  const hashHex = hashArray.map(b => b.toString(16).padStart(2, '0')).join('');

  const numericHash = BigInt('0x' + hashHex);
  const otp = (numericHash % 900000n + 100000n).toString();

  const history = JSON.parse(localStorage.getItem('otp_history') || '[]');
  if (history.includes(otp)) return generateUniqueOTP(userId);
  history.push(otp);
  if (history.length > 10000) history.shift();
  localStorage.setItem('otp_history', JSON.stringify(history));

  const record = {
    otp, userId, counter,
    hash: hashHex.substring(0, 16),
    issuedAt: new Date().toISOString(),
    expiresAt: new Date(Date.now() + 5 * 60 * 1000).toISOString(),
  };
  localStorage.setItem('otp_current', JSON.stringify(record));
  AuditLog.add(record);

  console.log(`%c🔐 OTP: ${otp}`, 'background: linear-gradient(135deg,#8B5CF6,#EC4899); color:white; padding:6px 14px; border-radius:8px; font-weight:bold; font-size:14px;');
  return record;
}

// ========== OTP Inputs ==========
const OTPInputs = {
  inputs: [],
  onComplete: null,
  init(cb) {
    this.inputs = Array.from(document.querySelectorAll('#otpInputs input'));
    this.onComplete = cb;
    this.inputs.forEach((inp, i) => {
      inp.addEventListener('input', (e) => {
        let v = e.target.value.replace(/\D/g, '');
        e.target.value = v;
        if (v && i < this.inputs.length - 1) this.inputs[i + 1].focus();
        e.target.classList.toggle('filled', !!v);
        if (this.inputs.every(x => x.value) && this.onComplete) this.onComplete(this.getValue());
      });
      inp.addEventListener('keydown', (e) => {
        if (e.key === 'Backspace' && !e.target.value && i > 0) {
          this.inputs[i - 1].focus();
          this.inputs[i - 1].value = '';
          this.inputs[i - 1].classList.remove('filled');
        }
        if (e.key === 'Enter') document.getElementById('verifyBtn').click();
      });
      inp.addEventListener('paste', (e) => {
        e.preventDefault();
        const text = (e.clipboardData || window.clipboardData).getData('text').replace(/\D/g, '');
        if (text.length >= 6) {
          this.inputs.forEach((inp, idx) => {
            inp.value = text[idx] || '';
            inp.classList.toggle('filled', !!inp.value);
          });
          this.inputs[5].focus();
          if (this.onComplete) this.onComplete(this.getValue());
        }
      });
    });
  },
  getValue() { return this.inputs.map(i => i.value).join(''); },
  clear() { this.inputs.forEach(i => { i.value = ''; i.classList.remove('filled'); }); },
  shake() {
    const wrap = document.getElementById('otpInputs');
    wrap.classList.add('shake');
    setTimeout(() => wrap.classList.remove('shake'), 500);
  },
};

// ========== Countdown ==========
const Countdown = {
  intervalId: null, seconds: 300, maxSeconds: 300, onExpire: null,
  start(seconds, onExpire) {
    this.stop();
    this.seconds = seconds;
    this.maxSeconds = seconds;
    this.onExpire = onExpire;
    this._render();
    this.intervalId = setInterval(() => {
      this.seconds--;
      this._render();
      if (this.seconds <= 0) { this.stop(); if (this.onExpire) this.onExpire(); }
    }, 1000);
  },
  stop() { if (this.intervalId) clearInterval(this.intervalId); this.intervalId = null; },
  _render() {
    const m = Math.floor(this.seconds / 60);
    const s = this.seconds % 60;
    const text = `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}`;
    const textEl = document.getElementById('countdownText');
    const circle = document.getElementById('countdownCircle');
    if (textEl) textEl.textContent = text;
    if (circle) {
      const circumference = 213.6;
      const ratio = this.seconds / this.maxSeconds;
      circle.style.strokeDashoffset = String(circumference * (1 - ratio));
      let color = '#10B981';
      if (this.seconds < 60) color = '#EF4444';
      else if (this.seconds < 180) color = '#F59E0B';
      circle.setAttribute('stroke', color);
      textEl.style.color = color;
    }
  },
};

// ========== Push Notifications ==========
const PushNotify = {
  permission: 'default',
  timers: {},
  async init() {
    if (!('Notification' in window)) return false;
    this.permission = Notification.permission;
    if (this.permission === 'default') {
      const req = async () => {
        try {
          const p = await Notification.requestPermission();
          this.permission = p;
          if (p === 'granted') {
            this.show('🔔 تم تفعيل الإشعارات', { body: 'سنذكرك عند اقتراب انتهاء صلاحية الرمز' });
          }
        } catch (e) { console.warn(e); }
        document.removeEventListener('click', req);
      };
      document.addEventListener('click', req, { once: true });
    }
    return this.permission === 'granted';
  },
  show(title, options = {}) {
    const opts = { dir: 'rtl', lang: 'ar', vibrate: [200, 100, 200], ...options };
    if (this.permission === 'granted') {
      try { return new Notification(title, opts); } catch (e) { console.warn(e); }
    }
    UI.toast(`${title}${opts.body ? ' — ' + opts.body : ''}`, 'info', 5000);
  },
  scheduleExpiryWarnings(record) {
    if (!record || !record.expiresAt) return;
    const expiresAt = new Date(record.expiresAt).getTime();
    const otpId = record.otp;
    if (this.timers[otpId]) {
      clearTimeout(this.timers[otpId].warn);
      clearTimeout(this.timers[otpId].expire);
    }
    const warnDelay = expiresAt - Date.now() - 60000;
    const expireDelay = expiresAt - Date.now();
    const warn = setTimeout(() => {
      this.show('⏰ الرمز على وشك الانتهاء', { body: 'يتبقى أقل من دقيقة — أكد الرمز الآن', requireInteraction: true });
    }, Math.max(warnDelay, 0));
    const expire = setTimeout(() => {
      this.show('❌ انتهت صلاحية الرمز', { body: 'اطلب رمزاً جديداً للمتابعة', requireInteraction: true });
      AuditLog.markExpired(otpId);
    }, Math.max(expireDelay, 0));
    this.timers[otpId] = { warn, expire };
  },
  cancelAll() {
    Object.values(this.timers).forEach(t => { clearTimeout(t.warn); clearTimeout(t.expire); });
    this.timers = {};
  },
};

  init() {
    const saved = localStorage.getItem(this.KEY) || 'auto';
    this.apply(saved);
    if (window.matchMedia) {
      window.matchMedia('(prefers-color-scheme: dark)').addEventListener('change', () => {
        if (localStorage.getItem(this.KEY) === 'auto') this.apply('auto');
      });
    }
    const btn = document.getElementById('themeToggleBtn');
    if (btn) btn.addEventListener('click', () => this.toggle());
    this._updateIcon();
  },
  apply(mode) {
    const isDark = mode === 'dark' || (mode === 'auto' && window.matchMedia('(prefers-color-scheme: dark)').matches);
    document.documentElement.classList.toggle('dark', isDark);
    localStorage.setItem(this.KEY, mode);
    this._updateIcon();
  },
  toggle() {
    const current = localStorage.getItem(this.KEY) || 'auto';
    const next = { auto: 'light', light: 'dark', dark: 'auto' }[current] || 'dark';
    this.apply(next);
    const labels = { light: '☀️ الوضع النهاري', dark: '🌙 الوضع الليلي', auto: '🔄 تلقائي حسب النظام' };
    UI.toast(labels[next], 'info', 2000);
  },
  _updateIcon() {
    const btn = document.getElementById('themeToggleBtn');
    if (!btn) return;
    const icon = btn.querySelector('i');
    const mode = localStorage.getItem(this.KEY) || 'auto';
    const isDark = document.documentElement.classList.contains('dark');
    if (mode === 'auto') icon.className = 'fa-solid fa-circle-half-stroke';
    else if (isDark) icon.className = 'fa-solid fa-sun';
    else icon.className = 'fa-solid fa-moon';
  },
};

// ========== Stats Dashboard ==========
const Stats = {
  chart: null,
  getMetrics() {
    const audit = JSON.parse(localStorage.getItem('otp_audit_log') || '[]');
    const counter = parseInt(localStorage.getItem('otp_counter') || '0', 10);
    const history = JSON.parse(localStorage.getItem('otp_history') || '[]');
    const verified = audit.filter(a => a.status === 'verified').length;
    const expired = audit.filter(a => a.status === 'expired').length;
    const pending = audit.filter(a => a.status === 'pending').length;
    const total = audit.length || 1;
    const successRate = ((verified / total) * 100).toFixed(1);
    const now = Date.now();
    const buckets = Array(24).fill(0);
    audit.forEach(e => {
      const h = Math.floor((now - new Date(e.issuedAt).getTime()) / 3600000);
      if (h >= 0 && h < 24) buckets[23 - h]++;
    });
    return { counter, historySize: history.length, auditSize: audit.length, verified, expired, pending, successRate, buckets, lastOtp: audit[audit.length - 1] };
  },
  render() {
    const m = this.getMetrics();
    const panel = document.getElementById('statsPanel');
    if (!panel) return;
    panel.innerHTML = `
      <div class="stats-header">
        <h3><i class="fa-solid fa-chart-line"></i> لوحة الإحصائيات</h3>
        <button onclick="Stats.render()" class="stats-refresh"><i class="fa-solid fa-rotate"></i></button>
      </div>
      <div class="stats-grid">
        <div class="stat-tile tile-purple">
          <div class="tile-icon"><i class="fa-solid fa-hashtag"></i></div>
          <div class="tile-body"><span class="tile-label">إجمالي الرموز</span><strong class="tile-value">${m.counter.toLocaleString('ar-EG')}</strong></div>
        </div>
        <div class="stat-tile tile-green">
          <div class="tile-icon"><i class="fa-solid fa-circle-check"></i></div>
          <div class="tile-body"><span class="tile-label">تم التحقق</span><strong class="tile-value">${m.verified}</strong></div>
        </div>
        <div class="stat-tile tile-red">
          <div class="tile-icon"><i class="fa-solid fa-clock-rotate-left"></i></div>
          <div class="tile-body"><span class="tile-label">منتهية</span><strong class="tile-value">${m.expired}</strong></div>
        </div>
        <div class="stat-tile tile-amber">
          <div class="tile-icon"><i class="fa-solid fa-hourglass-half"></i></div>
          <div class="tile-body"><span class="tile-label">معلّقة</span><strong class="tile-value">${m.pending}</strong></div>
        </div>
        <div class="stat-tile tile-pink">
          <div class="tile-icon"><i class="fa-solid fa-bullseye"></i></div>
          <div class="tile-body"><span class="tile-label">معدل النجاح</span><strong class="tile-value">${m.successRate}%</strong></div>
        </div>
        <div class="stat-tile tile-cyan">
          <div class="tile-icon"><i class="fa-solid fa-database"></i></div>
          <div class="tile-body"><span class="tile-label">في السجل</span><strong class="tile-value">${m.historySize}</strong></div>
        </div>
      </div>
      <div class="stats-chart-wrap">
        <h4>التوزيع على مدار 24 ساعة</h4>
        <canvas id="statsChart" height="120"></canvas>
      </div>
      ${m.lastOtp ? `
        <div class="stats-last">
          <span>آخر رمز:</span>
          <code>${m.lastOtp.otp.slice(0,2)}****${m.lastOtp.otp.slice(-2)}</code>
          <span class="badge-status ${m.lastOtp.status}">${ {verified:'تم التحقق', expired:'منتهي', pending:'معلّق'}[m.lastOtp.status] }</span>
        </div>
      ` : ''}
    `;
    this._renderChart(m.buckets);
  },
  _renderChart(data) {
    const canvas = document.getElementById('statsChart');
    if (!canvas) return;
    if (this.chart) this.chart.destroy();
    const ctx = canvas.getContext('2d');
    const gradient = ctx.createLinearGradient(0, 0, 0, 200);
    gradient.addColorStop(0, 'rgba(139, 92, 246, 0.8)');
    gradient.addColorStop(1, 'rgba(236, 72, 153, 0.1)');
    const labels = Array.from({ length: 24 }, (_, i) => {
      const h = (new Date().getHours() - 23 + i + 24) % 24;
      return `${String(h).padStart(2, '0')}:00`;
    });
    this.chart = new Chart(ctx, {
      type: 'line',
      data: {
        labels,
        datasets: [{
          data, borderColor: '#8B5CF6', backgroundColor: gradient,
          borderWidth: 2, tension: 0.4, fill: true, pointRadius: 3,
          pointBackgroundColor: '#EC4899', pointBorderColor: 'white', pointBorderWidth: 2,
        }],
      },
      options: {
        responsive: true, maintainAspectRatio: false,
        plugins: { legend: { display: false } },
        scales: {
          x: { grid: { display: false }, ticks: { font: { family: 'Cairo', size: 9 }, color: '#9ca3af', maxTicksLimit: 6 } },
          y: { beginAtZero: true, grid: { color: 'rgba(139, 92, 246, 0.08)' }, ticks: { font: { family: 'Cairo', size: 10 }, stepSize: 1 } },
        },
      },
    });
  },
  init() { this.render(); setInterval(() => this.render(), 8000); },
};

// ========== Verify Flow ==========
const Verify = {
  currentEmail: '', currentMobile: '',
  resendInterval: null,
  init() {
    OTPInputs.init((code) => this.tryVerify(code));
    document.getElementById('verifyBtn').addEventListener('click', () => {
      const code = OTPInputs.getValue();
      if (code.length !== 6) { OTPInputs.shake(); UI.toast('أدخل 6 أرقام كاملة', 'warn'); return; }
      this.tryVerify(code);
    });
    document.getElementById('resendBtn').addEventListener('click', () => this.resend());
  },
  async send(email, mobile) {
    this.currentEmail = email;
    this.currentMobile = mobile;
    const record = await generateUniqueOTP(email);
    document.getElementById('maskedContact').textContent = UI.maskContact(email, mobile);
    document.getElementById('demoOtpValue').textContent = record.otp;
    OTPInputs.clear();
    document.getElementById('screen-otp').classList.add('active');
    Countdown.start(300, () => {
      UI.toast('انتهت صلاحية الرمز — أعد الإرسال', 'error');
      OTPInputs.shake();
    });
    this._startResendTimer(30);
    UI.toast(`تم إرسال الرمز إلى ${UI.maskContact(email, mobile)}`, 'success');
    UI.updateCounter();
    PushNotify.scheduleExpiryWarnings(record);
    PushNotify.show('🔐 رمزك جاهز', { body: 'أدخل الرمز خلال 5 دقائق' });
  },
  tryVerify(code) {
    const stored = localStorage.getItem('otp_current');
    if (!stored) { OTPInputs.shake(); UI.toast('لا يوجد رمز نشط', 'error'); return; }
    const record = JSON.parse(stored);
    if (Date.now() > new Date(record.expiresAt).getTime()) {
      OTPInputs.shake();
      AuditLog.markExpired(record.otp);
      UI.toast('انتهت صلاحية الرمز', 'error');
      localStorage.removeItem('otp_current');
      return;
    }
    if (record.otp !== code) {
      OTPInputs.shake();
      AuditLog.markExpired(record.otp);
      UI.toast('الرمز غير صحيح', 'error');
      return;
    }
    AuditLog.markVerified(record.otp);
    localStorage.removeItem('otp_current');
    Countdown.stop();
    PushNotify.cancelAll();
    document.getElementById('metaIssued').textContent = new Date(record.issuedAt).toLocaleTimeString('ar-EG');
    document.getElementById('metaHash').textContent = record.hash || '—';
    document.getElementById('metaCounter').textContent = record.counter || '—';
    document.getElementById('screen-otp').classList.remove('active');
    document.getElementById('screen-success').classList.add('active');
    this._confetti();
    UI.toast('🎉 تم التحقق بنجاح', 'success');
    localStorage.setItem('geoedu_session', JSON.stringify({ email: this.currentEmail, mobile: this.currentMobile, verifiedAt: new Date().toISOString() }));
  },
  _startResendTimer(seconds) {
    const btn = document.getElementById('resendBtn');
    const timerEl = document.getElementById('resendTimer');
    btn.disabled = true;
    let s = seconds;
    timerEl.textContent = s;
    if (this.resendInterval) clearInterval(this.resendInterval);
    this.resendInterval = setInterval(() => {
      s--;
      timerEl.textContent = s;
      if (s <= 0) { clearInterval(this.resendInterval); btn.disabled = false; }
    }, 1000);
  },
  async resend() { await this.send(this.currentEmail, this.currentMobile); },
  cancel() {
    Countdown.stop();
    if (this.resendInterval) clearInterval(this.resendInterval);
    localStorage.removeItem('otp_current');
    document.getElementById('screen-otp').classList.remove('active');
    UI.toast('تم الإلغاء', 'info');
  },
  _confetti() {
    const container = document.getElementById('confetti');
    container.innerHTML = '';
    const colors = ['#8B5CF6', '#EC4899', '#F59E0B', '#06B6D4', '#10B981'];
    for (let i = 0; i < 60; i++) {
      const p = document.createElement('div');
      p.className = 'confetti-piece';
      p.style.left = `${Math.random() * 100}%`;
      p.style.background = colors[Math.floor(Math.random() * colors.length)];
      p.style.animationDelay = `${Math.random() * 0.8}s`;
      p.style.animationDuration = `${2 + Math.random() * 2}s`;
      container.appendChild(p);
      setTimeout(() => p.remove(), 5000);
    }
  },
};
